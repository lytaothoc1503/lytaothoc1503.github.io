          if (currentUser.shift > 2) return;

          btnTask.disabled = true;
          orderSub.textContent = 'Đang ghép nối dữ liệu tiếp thị...';
          orderSub.style.color = 'var(--gold-light)';

          let rolls = 0;
          const timer = setInterval(() => {
            const temp =
              demoOrders[Math.floor(Math.random() * demoOrders.length)];
            orderName.textContent = temp.name;
            orderComm.textContent =
              '+' + Number(temp.comm).toLocaleString('vi-VN') + ' đ';
            rolls++;

            if (rolls > 6) {
              clearInterval(timer);
              const lucky =
                demoOrders[Math.floor(Math.random() * demoOrders.length)];
              orderName.textContent = lucky.name;
              orderComm.textContent =
                '+' + Number(lucky.comm).toLocaleString('vi-VN') + ' đ';
              orderSub.textContent = 'Chúc mừng! Đơn hàng đã ghép thành công.';
              orderSub.style.color = 'var(--emerald-green)';

              currentUser.balance = (currentUser.balance || 0) + lucky.comm;
              currentUser.count += 1;

              // GHI NHẬT KÝ ĐƠN HÀNG
              let myOrdersLog =
                JSON.parse(
                  localStorage.getItem(
                    'xmagic_orders_log_' + currentUser.username
                  )
                ) || [];
              myOrdersLog.unshift({
                name: lucky.name,
                comm: lucky.comm,
                shift: currentUser.shift,
                orderIndex: currentUser.count,
                time:
                  new Date().toLocaleTimeString('vi-VN') +
                  ' - ' +
                  new Date().toLocaleDateString('vi-VN'),
              });
              localStorage.setItem(
                'xmagic_orders_log_' + currentUser.username,
                JSON.stringify(myOrdersLog)
              );

              if (currentUser.count >= 50) {
                if (currentUser.shift === 1) {
                  currentUser.shift = 2;
                  currentUser.count = 0;
                  alert(
                    'Chúc mừng bạn đã hoàn thành Lần 1 (50/50 đơn)! Bắt đầu tiến hành Lần 2.'
                  );
                } else {
                  currentUser.shift = 3;
                  alert('Bạn đã hoàn tất toàn bộ 2 lần gửi đơn hôm nay!');
                }
              }

              saveUserData();
              updateBalanceUI();
              renderTaskUI();
              renderOrdersLogUI();
              btnTask.disabled = false;
            }
          }, 80);
        });

        // LIÊN KẾT NGÂN HÀNG
        document
          .getElementById('link-bank-form')
          .addEventListener('submit', (e) => {
            e.preventDefault();
            currentUser.bank =
              document.getElementById('link-bank-select').value;
            currentUser.bankAcc =
              document.getElementById('link-bank-acc').value;
            currentUser.bankHolder = document
              .getElementById('link-bank-holder')
              .value.toUpperCase();

            saveUserData();
            checkLinkedBankUI();
            closeModal('bank-modal');
            alert('Liên kết tài khoản ngân hàng thành công!');
          });

        // ĐỔI MẬT KHẨU
        document
          .getElementById('change-pass-form')
          .addEventListener('submit', (e) => {
            e.preventDefault();
            const oldP = document.getElementById('old-pass').value;
            const newP = document.getElementById('new-pass').value;

            if (oldP !== currentUser.password) {
              alert('Mật khẩu hiện tại không chính xác!');
              return;
            }
            if (newP.length < 6) {
              alert('Mật khẩu mới phải từ 6 ký tự trở lên!');
              return;
            }

            currentUser.password = newP;
            saveUserData();
            closeModal('password-modal');
            alert('Đổi mật khẩu thành công!');
            document.getElementById('change-pass-form').reset();
          });

        // RÚT TIỀN
        document
          .getElementById('withdraw-sheet-form')
          .addEventListener('submit', (e) => {
            e.preventDefault();
            const money = parseInt(
              document.getElementById('withdraw-sheet-money').value
            );

            if (!currentUser.bankAcc) {
              alert(
                'Vui lòng liên kết tài khoản ngân hàng trước khi rút tiền!'
              );
              closeModal('withdraw-modal');
              openModal('bank-modal');
              return;
            }
            if (money > (currentUser.balance || 0)) {
              alert('Số dư ví không đủ để rút!');
              return;
            }
            if (money < 50000) {
              alert('Số tiền rút tối thiểu là 50.000 VNĐ!');
              return;
            }

            currentUser.balance -= money;
            saveUserData();
            updateBalanceUI();

            let withdraws =
              JSON.parse(localStorage.getItem('xmagic_db_withdraws')) || [];
            withdraws.push({
              time:
                new Date().toLocaleTimeString('vi-VN') +
                ' - ' +
                new Date().toLocaleDateString('vi-VN'),
              username: currentUser.username,
              bank: currentUser.bank,
              account:
                currentUser.bankAcc + ' (' + currentUser.bankHolder + ')',
              amount: money,
              status: 'Chờ duyệt',
            });
            localStorage.setItem(
              'xmagic_db_withdraws',
              JSON.stringify(withdraws)
            );

            closeModal('withdraw-modal');
            alert('Yêu cầu rút tiền thành công! Lệnh đang chờ xét duyệt.');
            document.getElementById('withdraw-sheet-form').reset();
          });

        // ĐĂNG XUẤT
        document
          .getElementById('btn-logout-full')
          .addEventListener('click', () => {
            if (confirm('Bạn có chắc chắn muốn đăng xuất?')) {
              localStorage.removeItem('xmagic_user');
              window.location.replace('auth.html');
            }
          });
      });

      function renderOrdersLogUI() {
        if (!currentUser) return;
        const myOrdersLog =
          JSON.parse(
            localStorage.getItem('xmagic_orders_log_' + currentUser.username)
          ) || [];
        const container = document.getElementById('orders-log-list');
        const counter = document.getElementById('logs-total-count');

        counter.textContent = `${myOrdersLog.length} đơn đã gửi`;

        if (myOrdersLog.length === 0) {
          container.innerHTML = `<p style="text-align:center; color:var(--text-dim); font-size:12.5px; padding:25px 0;">Chưa có dữ liệu gửi đơn nào được thực hiện.</p>`;
          return;
        }

        container.innerHTML = myOrdersLog
          .map(
            (item) => `
                <div class="logs-item">
                    <div class="logs-item-top">
                        <span style="color:#e2e8f0;">${item.name}</span>
                        <span style="color:var(--emerald-green);">+${Number(
                          item.comm
                        ).toLocaleString('vi-VN')} đ</span>
                    </div>
                    <div class="logs-item-time">
                        <span>Thời gian: ${item.time}</span>
                        <span style="color:var(--gold-light);">Lần ${
                          item.shift
                        } (Đơn #${item.orderIndex})</span>
                    </div>
                </div>
            `
          )
          .join('');
      }

      function renderUserInfo() {
        const name = currentUser.fullname || 'Thành viên';
        const genderPrefix = currentUser.gender
          ? `(${currentUser.gender})`
          : '';

        document.getElementById('user-display-name').textContent = name;
        document.getElementById(
          'profile-fullname'
        ).textContent = `${name} ${genderPrefix}`;
        document.getElementById('profile-sub-info').textContent =
          'Mã đại lý: ' + currentUser.username;
        document.getElementById('user-avatar-initial').textContent = name
          .charAt(0)
          .toUpperCase();
        document.getElementById('profile-vip-tag').textContent =
          currentUser.vipLevel || 'PRIORITY VIP 1';
      }

      function updateBalanceUI() {
        const balStr =
          Number(currentUser.balance || 0).toLocaleString('vi-VN') + ' đ';
        document.getElementById('card-balance').textContent = balStr;
        const sheetBal = document.getElementById('withdraw-sheet-bal');
        if (sheetBal) sheetBal.value = balStr;
      }

      function checkLinkedBankUI() {
        const statusEl = document.getElementById('linked-bank-status');
        if (currentUser.bankAcc) {
          statusEl.textContent =
            currentUser.bank + ' (...' + currentUser.bankAcc.slice(-4) + ')';
          statusEl.style.color = 'var(--emerald-green)';
          statusEl.style.fontWeight = 'bold';
        } else {
          statusEl.textContent = 'Chưa liên kết';
          statusEl.style.color = 'var(--text-dim)';
        }
      }

      function saveUserData() {
        localStorage.setItem('xmagic_user', JSON.stringify(currentUser));
        let users = JSON.parse(localStorage.getItem('xmagic_db_users')) || [];
        const idx = users.findIndex((u) => u.username === currentUser.username);
        if (idx !== -1) {
          users[idx] = currentUser;
          localStorage.setItem('xmagic_db_users', JSON.stringify(users));
        }
      }

      function switchTab(tabId) {
        document
          .querySelectorAll('.tab-pane')
          .forEach((p) => p.classList.remove('active'));
        document
          .querySelectorAll('.nav-btn')
          .forEach((b) => b.classList.remove('active'));

        const target = document.getElementById('tab-' + tabId);
        if (target) target.classList.add('active');

        const indexMap = { home: 0, task: 1, orders: 2, profile: 3 };
        const btns = document.querySelectorAll('.nav-btn');
        if (btns[indexMap[tabId]])
          btns[indexMap[tabId]].classList.add('active');

        if (tabId === 'orders') {
          renderOrdersLogUI();
        }
      }

      function openModal(modalId) {
        if (modalId === 'history-modal') {
          renderHistoryUI();
        }
        const el = document.getElementById(modalId);
        if (el) el.classList.add('active');
      }

      function closeModal(modalId) {
        const el = document.getElementById(modalId);
        if (el) el.classList.remove('active');
      }

      function renderHistoryUI() {
        const withdraws =
          JSON.parse(localStorage.getItem('xmagic_db_withdraws')) || [];
        const myHistory = withdraws.filter(
          (w) => w.username === currentUser.username
        );
        const container = document.getElementById('history-list-content');

        if (myHistory.length === 0) {
          container.innerHTML = `<p style="text-align:center; color:var(--text-dim); padding:20px;">Chưa có giao dịch rút tiền nào.</p>`;
          return;
        }

        container.innerHTML = myHistory
          .map(
            (item) => `
                <div style="background: rgba(0,0,0,0.35); border-bottom: 1px solid var(--border-subtle); padding: 12px; border-radius: 8px; margin-bottom: 8px;">
                    <div style="display: flex; justify-content: space-between; font-weight: bold;">
                        <span style="color:#ffffff;">-${Number(
                          item.amount
                        ).toLocaleString('vi-VN')} đ</span>
                        <span style="color: var(--gold-light); font-size: 12px;">${
                          item.status
                        }</span>
                    </div>
                    <div style="font-size: 11.5px; color: var(--text-dim); margin-top: 4px;">
                        Ngân hàng: ${item.bank} - ${item.account}<br>
                        Thời gian: ${item.time}
                    </div>
                </div>
            `
          )
          .join('');
      }
