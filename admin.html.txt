    <script>
      let cachedUsersHash = '';
      let cachedWithdrawHash = '';
      let globalUsers = [];

      // HÀM ĐỒNG BỘ REAL-TIME KHÔNG CẦN F5
      function syncEngine() {
        globalUsers = JSON.parse(localStorage.getItem('xmagic_db_users')) || [];
        const withdraws =
          JSON.parse(localStorage.getItem('xmagic_db_withdraws')) || [];

        const currentUsersHash = JSON.stringify(globalUsers);
        const currentWithdrawHash = JSON.stringify(withdraws);

        if (
          currentUsersHash !== cachedUsersHash ||
          currentWithdrawHash !== cachedWithdrawHash
        ) {
          cachedUsersHash = currentUsersHash;
          cachedWithdrawHash = currentWithdrawHash;
          renderDashboard(globalUsers, withdraws);
        }
      }

      // TẢI CẤU HÌNH HỆ THỐNG LÊN GIAO DIỆN
      function loadSystemConfig() {
        const cfg = JSON.parse(localStorage.getItem('xmagic_sys_config')) || {
          telegram: 'https://telegram.org',
          hotline: '0988.xxx.xxx',
        };
        document.getElementById('cfg-telegram').value = cfg.telegram;
        document.getElementById('cfg-hotline').value = cfg.hotline;
      }

      // LƯU CẤU HÌNH THẬT
      document
        .getElementById('system-config-form')
        .addEventListener('submit', (e) => {
          e.preventDefault();
          const newCfg = {
            telegram: document.getElementById('cfg-telegram').value.trim(),
            hotline: document.getElementById('cfg-hotline').value.trim(),
          };
          localStorage.setItem('xmagic_sys_config', JSON.stringify(newCfg));
          alert(
            'Đã cập nhật và lưu cấu hình hệ thống thành công! Toàn bộ khách hàng sẽ thấy thông tin mới ngay lập tức.'
          );
        });

      // VẼ DỮ LIỆU BẢNG VÀ THỐNG KÊ
      function renderDashboard(users, withdraws) {
        document.getElementById('stat-total-users').textContent = users.length;

        let totalFunds = 0;
        let totalTasksCompleted = 0;
        users.forEach((u) => {
          totalFunds += Number(u.balance || 0);
          totalTasksCompleted +=
            Number(u.count || 0) + (Number(u.shift || 1) - 1) * 50;
        });
        document.getElementById('stat-total-funds').textContent =
          totalFunds.toLocaleString('vi-VN') + ' đ';
        document.getElementById('stat-total-tasks').textContent =
          totalTasksCompleted;

        const pendingList = withdraws.filter(
          (w) => w.status === 'Chờ duyệt' || w.status === 'Chờ thanh khoản'
        );
        document.getElementById('stat-pending-withdraw').textContent =
          pendingList.length;

        renderUserTable(users);

        // BẢNG RÚT TIỀN
        const financeBody = document.getElementById('finance-table-body');
        financeBody.innerHTML = '';
        if (withdraws.length === 0) {
          financeBody.innerHTML = `<tr><td colspan="7" style="text-align:center; padding:20px; color:var(--text-dim);">Chưa ghi nhận yêu cầu rút tiền nào.</td></tr>`;
        } else {
          withdraws.forEach((w, idx) => {
            const tr = document.createElement('tr');
            const isPending =
              w.status === 'Chờ duyệt' || w.status === 'Chờ thanh khoản';
            tr.innerHTML = `
                        <td>${w.time}</td>
                        <td style="color:var(--gold-light);">${w.username}</td>
                        <td><strong>${w.bank}</strong></td>
                        <td style="font-family:monospace;">${w.account}</td>
                        <td style="color:var(--emerald-green); font-weight:700;">${Number(
                          w.amount
                        ).toLocaleString('vi-VN')} đ</td>
                        <td><span style="color:${
                          isPending
                            ? 'var(--gold-light)'
                            : w.status === 'Thành công'
                            ? 'var(--emerald-green)'
                            : 'var(--red-alert)'
                        }; font-weight:600;">${w.status}</span></td>
                        <td>
                            ${
                              isPending
                                ? `
                                <button class="btn-act btn-green" onclick="resolveWithdraw(${idx}, true)">Duyệt</button>
                                <button class="btn-act btn-red" onclick="resolveWithdraw(${idx}, false)">Từ Chối</button>
                            `
                                : `<span style="color:var(--text-dim);">Đã xử lý</span>`
                            }
                        </td>
                    `;
            financeBody.appendChild(tr);
          });
        }

        // BẢNG NHẬT KÝ ĐƠN HÀNG
        const ordersBody = document.getElementById('orders-table-body');
        ordersBody.innerHTML = '';
        let aggregatedLogs = [];
        users.forEach((u) => {
          const logs =
            JSON.parse(
              localStorage.getItem('xmagic_orders_log_' + u.username)
            ) || [];
          logs.forEach((l) => aggregatedLogs.push({ ...l, user: u.username }));
        });

        if (aggregatedLogs.length === 0) {
          ordersBody.innerHTML = `<tr><td colspan="5" style="text-align:center; padding:20px; color:var(--text-dim);">Chưa có đơn tiếp thị nào được gửi.</td></tr>`;
        } else {
          aggregatedLogs.slice(0, 50).forEach((l) => {
            const tr = document.createElement('tr');
            tr.innerHTML = `
                        <td>${l.time}</td>
                        <td style="color:var(--gold-light);">${l.user}</td>
                        <td>${l.name}</td>
                        <td style="color:var(--emerald-green); font-weight:700;">+${Number(
                          l.comm
                        ).toLocaleString('vi-VN')} đ</td>
                        <td>Ca ${l.shift} (#${l.orderIndex})</td>
                    `;
            ordersBody.appendChild(tr);
          });
        }
      }

      // VẼ BẢNG USER (KÈM LỌC TÌM KIẾM)
      function renderUserTable(usersToRender) {
        const userBody = document.getElementById('users-table-body');
        userBody.innerHTML = '';

        if (usersToRender.length === 0) {
          userBody.innerHTML = `<tr><td colspan="7" style="text-align:center; padding:20px; color:var(--text-dim);">Không tìm thấy thành viên nào.</td></tr>`;
          return;
        }

        usersToRender.forEach((u) => {
          const realIndex = globalUsers.findIndex(
            (item) => item.username === u.username
          );
          const isBlocked = u.status === 'blocked';
          const tr = document.createElement('tr');
          tr.innerHTML = `
                    <td><strong>${
                      u.fullname || 'Thành viên'
                    }</strong> <small style="color:var(--text-dim)">(${
            u.gender || 'Chưa chọn'
          })</small></td>
                    <td style="color:var(--gold-light);">${u.username}</td>
                    <td><span style="font-size:11px; color:var(--gold-light);">${
                      u.vipLevel || 'PRIORITY VIP 1'
                    }</span></td>
                    <td style="color:var(--emerald-green); font-weight:700;">${Number(
                      u.balance || 0
                    ).toLocaleString('vi-VN')} đ</td>
                    <td>Ca ${u.shift || 1} (${u.count || 0}/50)</td>
                    <td><span style="color:${
                      isBlocked ? 'var(--red-alert)' : 'var(--emerald-green)'
                    }; font-weight:600;">${
            isBlocked ? 'ĐÃ KHÓA' : 'HOẠT ĐỘNG'
          }</span></td>
                    <td>
                        <button class="btn-act btn-gold" onclick="openEditModal(${realIndex})">⚙️ Quản Trị</button>
                    </td>
                `;
          userBody.appendChild(tr);
        });
      }

      // TÌM KIẾM KHÁCH HÀNG TỨC THÌ
      function filterUsers() {
        const keyword = document
          .getElementById('user-search-input')
          .value.toLowerCase()
          .trim();
        if (keyword === '') {
          renderUserTable(globalUsers);
          return;
        }
        const filtered = globalUsers.filter(
          (u) =>
            (u.fullname && u.fullname.toLowerCase().includes(keyword)) ||
            (u.username && u.username.toLowerCase().includes(keyword))
        );
        renderUserTable(filtered);
      }

      // MỞ POPUP QUẢN TRỊ THÀNH VIÊN
      function openEditModal(index) {
        const user = globalUsers[index];
        if (!user) return;

        document.getElementById('edit-user-index').value = index;
        document.getElementById('edit-user-fullname').value =
          user.fullname || '';
        document.getElementById('edit-user-username').value =
          user.username || '';
        document.getElementById('edit-user-password').value =
          user.password || '';
        document.getElementById('edit-user-vip').value =
          user.vipLevel || 'PRIORITY VIP 1';
        document.getElementById('edit-user-balance').value = user.balance || 0;
        document.getElementById('edit-user-count').value = user.count || 0;
        document.getElementById('edit-user-shift').value = user.shift || 1;
        document.getElementById('edit-user-status').value =
          user.status || 'active';

        document.getElementById('edit-user-modal').classList.add('active');
      }

      function closeEditModal() {
        document.getElementById('edit-user-modal').classList.remove('active');
      }

      // LƯU CẬP NHẬT THÀNH VIÊN TỪ MODAL
      document
        .getElementById('save-user-form')
        .addEventListener('submit', (e) => {
          e.preventDefault();
          const index = document.getElementById('edit-user-index').value;
          let user = globalUsers[index];
          if (!user) return;

          const oldUsername = user.username;

          user.fullname = document
            .getElementById('edit-user-fullname')
            .value.trim();
          user.username = document
            .getElementById('edit-user-username')
            .value.trim();
          user.password = document
            .getElementById('edit-user-password')
            .value.trim();
          user.vipLevel = document.getElementById('edit-user-vip').value;
          user.balance = parseInt(
            document.getElementById('edit-user-balance').value
          );
          user.count = parseInt(
            document.getElementById('edit-user-count').value
          );
          user.shift = parseInt(
            document.getElementById('edit-user-shift').value
          );
          user.status = document.getElementById('edit-user-status').value;

          // Đồng bộ phiên nếu người này đang mở app
          let curUser = JSON.parse(localStorage.getItem('xmagic_user'));
          if (curUser && curUser.username === oldUsername) {
            localStorage.setItem('xmagic_user', JSON.stringify(user));
          }

          localStorage.setItem('xmagic_db_users', JSON.stringify(globalUsers));
          closeEditModal();
          syncEngine();
          alert(
            'Đã cập nhật toàn bộ quyền hạn và thông tin cho đại lý này thành công!'
          );
        });

      // XÓA TÀI KHOẢN VĨNH VIỄN
      function deleteUserPermanently() {
        const index = document.getElementById('edit-user-index').value;
        if (
          confirm(
            'CẢNH BÁO: Bạn có chắc chắn muốn xóa vĩnh viễn tài khoản này khỏi hệ thống?'
          )
        ) {
          globalUsers.splice(index, 1);
          localStorage.setItem('xmagic_db_users', JSON.stringify(globalUsers));
          closeEditModal();
          syncEngine();
        }
      }

      // DUYỆT / TỪ CHỐI RÚT TIỀN (HOÀN TIỀN NGAY LẬP TỨC NẾU TỪ CHỐI)
      function resolveWithdraw(index, isApproved) {
        let withdraws =
          JSON.parse(localStorage.getItem('xmagic_db_withdraws')) || [];
        const item = withdraws[index];
        if (!item) return;

        if (isApproved) {
          item.status = 'Thành công';
        } else {
          item.status = 'Bị từ chối';
          const uIdx = globalUsers.findIndex(
            (u) => u.username === item.username
          );
          if (uIdx !== -1) {
            globalUsers[uIdx].balance =
              (globalUsers[uIdx].balance || 0) + item.amount;
            localStorage.setItem(
              'xmagic_db_users',
              JSON.stringify(globalUsers)
            );

            let curUser = JSON.parse(localStorage.getItem('xmagic_user'));
            if (curUser && curUser.username === item.username) {
              curUser.balance = globalUsers[uIdx].balance;
              localStorage.setItem('xmagic_user', JSON.stringify(curUser));
            }
          }
        }

        localStorage.setItem('xmagic_db_withdraws', JSON.stringify(withdraws));
        syncEngine();
      }

          event.currentTarget.textContent;
      }
